## 確認過的 Model(權威來源;實作若與此牴觸,是計畫的 bug —— 停下來反映,不要就地改模型)

### Entities

- `AppPrefs { soundOn }` —— 全域,獨立 storage key
- 每個模式都有 `setups[]` 與 `activeSetupId`
- 扭蛋機 2D:`Prize { id, name, count, rarity }` → `Capsule { prizeId, drawn }`
- 扭蛋機 3D:`Prize { id, name, count, rarity }` → `Capsule { prizeId, drawn, color }`
- 一番賞:`IchibanPrize { id, name, tier(A~G), count }` + `lastOnePrize` → `Ticket { prizeId, drawn }`
- 阿彌陀籤:`Player { id, name, color }` + `GhostPrize { id, name, count }` → `Ladder`(每局現場產生,不存)
- 大亂鬥:`Fighter { id, name, color, count }` + `items{attack,speed,giant,bomb}` → `Combatant { fighterId, alive }`(不存)

### Cardinality

- `Prize` 1:N `Capsule`、`IchibanPrize` 1:N `Ticket`、`GhostPrize` 1:N 底部格子、
  `Fighter` 1:N `Combatant` —— 全部由 `count` 決定,同一個形狀
- `Player` 1:1 梯子上的一條線
- **`Fighter` 就是隊伍**,`count` 就是隊伍人數。同 `fighterId` 的 `Combatant` 不互相擊飛,
  最後還有人活著的 `fighterId` 獲勝

### Seen vs stored

- 3D 看到球體裡五顏六色的蛋 → **顏色是每顆蛋自己隨機挑的**,跟獎項、跟稀有度都無關
- 一番賞看到「A賞 PVC模型」→ 存的是 `tier: 'A'` 欄位,**不是名字前綴**
- 阿彌陀籤看到大家像賽跑有快有慢 → **誰拿到什麼在起跑前梯子就決定好了**,快慢純粹是演出
- 阿彌陀籤底部補的「銘謝惠顧」空籤是**算出來的**,不存進設定
- 大亂鬥是**唯一**結果不預先決定的模式

### 最可能做錯的四個(以下四句都是錯的)

1. 「一番賞的賞別就是稀有度」 —— **一番賞沒有稀有度,扭蛋機沒有賞別**
2. 「大亂鬥的隊伍是一個獨立實體」 —— 沒有 `Team`,隊伍就是同一筆 `Fighter`
3. 「3D 扭蛋機的蛋色由稀有度決定」 —— 只有 2D 那台是。3D 每顆蛋隨機上色
4. 「阿彌陀籤誰先跑到終點誰拿頭獎」 —— 名次跟獎項完全無關

`buildLadder` 的 `rungs` 中 `left` 表示這根橫線連接 `left` 與 `left + 1` 兩條直線;
同一 `row` 不得有相鄰的橫線(否則走法會產生歧義)。

function seeded(seed) {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6D2B79F5) >>> 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;

  const ladder = { lanes: 4, rows: 3, rungs: [] };
  assert.deepEqual([0, 1, 2, 3].map(i => walk(ladder, i)), [0, 1, 2, 3]);

test('一根橫線就是把相鄰兩條交換', () => {
  const ladder = { lanes: 4, rows: 3, rungs: [{ row: 1, left: 1 }] };
  assert.deepEqual([0, 1, 2, 3].map(i => walk(ladder, i)), [0, 2, 1, 3]);

test('兩根不同列的橫線依序套用', () => {
  const ladder = { lanes: 3, rows: 4, rungs: [{ row: 0, left: 0 }, { row: 2, left: 1 }] };
  // 起點 0:先換到 1,再換到 2
  assert.equal(walk(ladder, 0), 2);
  // 起點 1:先換到 0,之後不再動
  assert.equal(walk(ladder, 1), 0);

  assert.equal(walk(ladder, 2), 1);

    const ladder = buildLadder({ lanes, rng: seeded(seed) });
    const ends = Array.from({ length: lanes }, (_, i) => walk(ladder, i));
    assert.equal(new Set(ends).size, lanes, `seed ${seed} 的終點出現重複`);
  }

test('buildLadder 不產生同一列相鄰的橫線', () => {
  for (let seed = 1; seed <= 40; seed++) {
    const ladder = buildLadder({ lanes: 6, rng: seeded(seed) });
    const byRow = new Map();
    for (const rung of ladder.rungs) {
      if (!byRow.has(rung.row)) byRow.set(rung.row, []);
      byRow.get(rung.row).push(rung.left);
    }
    for (const [row, lefts] of byRow) {

// 阿彌陀籤(鬼腳圖)。誰拿到什麼由梯子決定,跟跑得快慢完全無關。
import { newId } from '../../shared/js/ids.js';
import { expand } from '../../shared/js/roster.js';

export const EMPTY_SLOT_NAME = '銘謝惠顧';

const PALETTE = [
  '#FF8D70', '#8CC9FF', '#9BE7C4', '#FFD479', '#C4A2FF',
  '#FF9DB5', '#7FD8C9', '#FFB37A', '#A8B8FF', '#F2A5D1',
];

export function createPlayer({ name = '玩家', color = null } = {}) {
  return { id: newId('pl'), name, color: color ?? PALETTE[Math.floor(Math.random() * PALETTE.length)] };
}

export function pickColor(existing) {
  const used = new Set(existing.map(p => p.color));
  return PALETTE.find(c => !used.has(c)) ?? PALETTE[existing.length % PALETTE.length];
}

export function createGhostPrize({ name = '新獎項', count = 1 } = {}) {
  return { id: newId('gp'), name, count: Math.max(0, Math.floor(count)) };
}

// 底部格子數恆等於玩家數:不足補空籤,超過只取前面幾個。
// 補出來的空籤是算的,不存進設定。
export function bottomSlots(prizes, playerCount) {
  const filled = expand(prizes, prize => ({ prizeId: prize.id, name: prize.name }));
  const slots = filled.slice(0, playerCount);
  while (slots.length < playerCount) slots.push({ prizeId: null, name: EMPTY_SLOT_NAME });
  return slots;
}

export function buildLadder({ lanes, rows = Math.max(6, lanes * 2), rng = Math.random }) {
  const rungs = [];
  for (let row = 0; row < rows; row++) {
    let left = 0;
    while (left < lanes - 1) {
      if (rng() < 0.45) {
        rungs.push({ row, left });
        left += 2;   // 跳過一條,避免同一列出現相鄰橫線(會讓走法產生歧義)
      } else {
        left += 1;
      }
    }
  }
  return { lanes, rows, rungs };
}

export function walk(ladder, startLane) {
  let lane = startLane;
  for (let row = 0; row < ladder.rows; row++) {
    const here = ladder.rungs.find(r => r.row === row && (r.left === lane || r.left + 1 === lane));
    if (!here) continue;
    lane = here.left === lane ? lane + 1 : lane - 1;
  }
  return lane;
}

export function assign(ladder, players, slots) {
  return players.map((player, lane) => {
    const slotIndex = walk(ladder, lane);
    return { playerId: player.id, slotIndex, slot: slots[slotIndex] };
  });
}
```

// 把 walk 的過程展開成逐列的座標,給動畫沿著畫。
// 這支不決定任何結果 —— 它的終點必定等於 walk。
export function pathOf(ladder, startLane) {
  const path = [{ row: -1, lane: startLane }];
  let lane = startLane;
  for (let row = 0; row < ladder.rows; row++) {
    const here = ladder.rungs.find(r => r.row === row && (r.left === lane || r.left + 1 === lane));
    if (here) {
      lane = here.left === lane ? lane + 1 : lane - 1;
      path.push({ row, lane });
    }
  }
  path.push({ row: ladder.rows, lane });
  return path;
}
```

## 已知驗不到的範圍

寫進計畫是為了讓執行者不要誤以為自己驗過了:

- **自動化瀏覽器的分頁是背景狀態、rAF 為 0 fps。** 3D 場景與物理模擬的實際觀感在開發過程中**無法驗證**,
  只能驗邏輯與狀態。Task 12 與 Task 15 的「手動確認」必須由人在前景分頁執行。
- **Safari / WebKit 與真實行動裝置沒有任何自動化覆蓋。** 既有的 2D 扭蛋機在 iOS 上就出現過
  「Reduce Motion 導致演出被略過」與「音效沒聲音」兩個只在 iOS 上發生的問題,
  三個新模式同樣有這個風險。
- **物理手感**(撞擊夠不夠爽、節奏會不會拖)只能靠人玩。測試只能保證規則正確,不能保證好玩。
