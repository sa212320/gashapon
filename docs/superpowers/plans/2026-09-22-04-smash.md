// 大亂鬥的物理。這是唯一結果不預先決定的模式 —— 誰贏由碰撞撒出來。
// 同隊(同 fighterId)之間只做位置分離,不施加任何衝量;不同隊才會互相擊飛。
import { expand } from '../../shared/js/roster.js';
import { newId } from '../../shared/js/ids.js';

const BASE_RADIUS = 14;
const RESTITUTION = 1.15;
const FRICTION = 0.995;

export function createWorld({ fighters, arenaRadius, rng = Math.random }) {
  const combatants = expand(fighters, fighter => {
    const angle = rng() * Math.PI * 2;
    const dist = Math.sqrt(rng()) * arenaRadius * 0.7;
    return {
      id: newId('c'),
      fighterId: fighter.id,
      x: Math.cos(angle) * dist,
      y: Math.sin(angle) * dist,
      vx: (rng() - 0.5) * 60,
      vy: (rng() - 0.5) * 60,
      radius: BASE_RADIUS,
      mass: 1,
      alive: true,
      buffs: { attack: 0, speed: 0, giant: 0 },
    };
  });
  return { arenaRadius, combatants, bombs: [], items: [], time: 0 };
}

export function aliveOf(world) {
  return world.combatants.filter(c => c.alive);
}

export function winnerTeam(world) {
  const teams = new Set(aliveOf(world).map(c => c.fighterId));
  return teams.size === 1 ? [...teams][0] : null;
}

export function step(world, dt) {
  const next = world.combatants.map(c => ({ ...c, buffs: { ...c.buffs } }));

  for (const c of next) {
    if (!c.alive) continue;
    c.x += c.vx * dt;
    c.y += c.vy * dt;
    c.vx *= FRICTION;
    c.vy *= FRICTION;
  }

  for (let i = 0; i < next.length; i++) {
    for (let j = i + 1; j < next.length; j++) {
      const a = next[i];
      const b = next[j];
      if (!a.alive || !b.alive) continue;

      const dx = b.x - a.x;
      const dy = b.y - a.y;
      const dist = Math.hypot(dx, dy) || 0.0001;
      const overlap = a.radius + b.radius - dist;
      if (overlap <= 0) continue;

      const nx = dx / dist;
      const ny = dy / dist;

      // 先分離 —— 同隊與不同隊都要做,不然會互相穿透
      const push = overlap / 2;
      a.x -= nx * push;
      a.y -= ny * push;
      b.x += nx * push;
      b.y += ny * push;

      // 同隊到此為止:不施加任何衝量
      if (a.fighterId === b.fighterId) continue;

      const rvx = b.vx - a.vx;
      const rvy = b.vy - a.vy;
      const along = rvx * nx + rvy * ny;
      if (along > 0) continue;

      const power = RESTITUTION * (1 + (a.buffs.attack ?? 0) * 0.6);
      const impulse = (-(1 + power) * along) / (1 / a.mass + 1 / b.mass);
      a.vx -= (impulse * nx) / a.mass;
      a.vy -= (impulse * ny) / a.mass;
      b.vx += (impulse * nx) / b.mass;
      b.vy += (impulse * ny) / b.mass;
    }
  }

  for (const c of next) {
    if (c.alive && Math.hypot(c.x, c.y) > world.arenaRadius + c.radius) c.alive = false;
  }

  return { ...world, combatants: next, time: world.time + dt };
}
```

// 四種道具。每一支都是 world → world 的純轉換,所以拾取範圍與爆炸衰減可以直接斷言。
// 注意:這些不塞進 physics.js 的 step(),那會讓物理測試(世界裡沒有 items)變紅。
import { newId } from '../../shared/js/ids.js';

export const ITEM_TYPES = ['attack', 'speed', 'giant', 'bomb'];

const ITEM_RADIUS = 18;
const BOMB_FUSE = 2.5;
const BLAST_RADIUS = 90;
const BLAST_POWER = 340;
const GIANT_SIZE = 1.6;
const GIANT_MASS = 4;

function scatter(world, rng) {
  const angle = rng() * Math.PI * 2;
  const dist = Math.sqrt(rng()) * world.arenaRadius * 0.8;
  return { x: Math.cos(angle) * dist, y: Math.sin(angle) * dist };
}

export function spawnItem(world, type, rng = Math.random) {
  const { x, y } = scatter(world, rng);
  return { ...world, items: [...(world.items ?? []), { id: newId('it'), type, x, y, radius: ITEM_RADIUS }] };
}

export function spawnBomb(world, rng = Math.random) {
  const { x, y } = scatter(world, rng);
  return { ...world, bombs: [...(world.bombs ?? []), { id: newId('bo'), x, y, fuse: BOMB_FUSE }] };
}

export function applyPickups(world) {
  const combatants = world.combatants.map(c => ({ ...c, buffs: { ...c.buffs } }));
  const taken = new Set();

  for (const item of world.items ?? []) {
    const who = combatants.find(c =>
      c.alive && Math.hypot(item.x - c.x, item.y - c.y) <= c.radius + item.radius);
    if (!who) continue;
    taken.add(item.id);

    if (item.type === 'attack') who.buffs.attack = (who.buffs.attack ?? 0) + 1;
    if (item.type === 'speed') who.buffs.speed = (who.buffs.speed ?? 0) + 1;
    if (item.type === 'giant') {
      who.buffs.giant = (who.buffs.giant ?? 0) + 1;
      who.radius *= GIANT_SIZE;
      who.mass *= GIANT_MASS;   // 變重才是真的抗擊飛 —— 物理那邊的衝量會除以 mass
    }
  }

  return { ...world, combatants, items: (world.items ?? []).filter(i => !taken.has(i.id)) };
}

export function explodeBombs(world, dt) {
  const combatants = world.combatants.map(c => ({ ...c }));
  const left = [];

  for (const bomb of world.bombs ?? []) {
    const fuse = bomb.fuse - dt;
    if (fuse > 0) { left.push({ ...bomb, fuse }); continue; }

    for (const c of combatants) {
      if (!c.alive) continue;
      const dx = c.x - bomb.x;
      const dy = c.y - bomb.y;
      const dist = Math.hypot(dx, dy);
      if (dist > BLAST_RADIUS) continue;
      const falloff = 1 - dist / BLAST_RADIUS;
      const nx = dist === 0 ? 1 : dx / dist;
      const ny = dist === 0 ? 0 : dy / dist;
      c.vx += (nx * BLAST_POWER * falloff) / c.mass;
      c.vy += (ny * BLAST_POWER * falloff) / c.mass;
    }
  }

  return { ...world, combatants, bombs: left };
}
```

## 已知驗不到的範圍

寫進計畫是為了讓執行者不要誤以為自己驗過了:

- **自動化瀏覽器的分頁是背景狀態、rAF 為 0 fps。** 3D 場景與物理模擬的實際觀感在開發過程中**無法驗證**,
  只能驗邏輯與狀態。`gashapon3d` 計畫 Task 2 與 `smash` 計畫 Task 3 的「手動確認」必須由人在前景分頁執行。
- **Safari / WebKit 與真實行動裝置沒有任何自動化覆蓋。** 既有的 2D 扭蛋機在 iOS 上就出現過
  「Reduce Motion 導致演出被略過」與「音效沒聲音」兩個只在 iOS 上發生的問題,
  三個新模式同樣有這個風險。
- **物理手感**(撞擊夠不夠爽、節奏會不會拖)只能靠人玩。測試只能保證規則正確,不能保證好玩。
